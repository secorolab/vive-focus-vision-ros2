cmake_minimum_required(VERSION 3.16)
project(vr C CXX) # C is required: rosidl generates C sources for the message types

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rosidl_default_generators REQUIRED)

# No standard gaze message exists in ROS 2, so this package generates the one type it needs.
rosidl_generate_interfaces(${PROJECT_NAME}
  msg/EyeGaze.msg
  DEPENDENCIES std_msgs geometry_msgs
)

# The wrapper is a plain CMake package colcon discovers; it brings the shared secorolab KDL.
find_package(mj_kdl_wrapper REQUIRED)

# Same user cache mj_kdl_wrapper fetches into, so both link one MuJoCo. No system paths are
# searched: a stray /opt/mujoco-3.8.0 would silently mismatch the wrapper's ABI.
set(VR_MUJOCO_DIR "$ENV{HOME}/.cache/mj_kdl_wrapper/mujoco-3.9.0"
    CACHE PATH "MuJoCo install root (include/ and lib/)")
find_library(MUJOCO_LIBRARY mujoco PATHS "${VR_MUJOCO_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h PATHS "${VR_MUJOCO_DIR}/include" NO_DEFAULT_PATH
          REQUIRED)
message(STATUS "MuJoCo: ${MUJOCO_LIBRARY}")

# --- reusable pieces -----------------------------------------------------------------------
# BodyPosePublisher is the seam that lets an existing mj_kdl_wrapper application stream itself
# to the headset without running vr::SceneNode at all.
add_library(vr_scene SHARED
  src/body_pose_publisher.cpp
  src/gltf_writer.cpp
  src/geom_mesh.cpp
)
target_include_directories(vr_scene PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
  "${MUJOCO_INCLUDE_DIR}"
)
target_link_libraries(vr_scene PUBLIC
  "${MUJOCO_LIBRARY}"
  rclcpp::rclcpp
  ${geometry_msgs_TARGETS}
  ${std_msgs_TARGETS}
)

# --- components ----------------------------------------------------------------------------
add_library(vr_components SHARED
  src/components/scene_node.cpp
  src/components/input_node.cpp
)
# Only the components depend on mj_kdl_wrapper; vr_scene stays MuJoCo-only so an application
# can link BodyPosePublisher without pulling in KDL.
target_link_libraries(vr_components PUBLIC
  vr_scene
  mj_kdl_wrapper::mj_kdl_wrapper
  rclcpp::rclcpp
  rclcpp_components::component
  ${builtin_interfaces_TARGETS}
  ${geometry_msgs_TARGETS}
  ${sensor_msgs_TARGETS}
  ${std_msgs_TARGETS}
  ${std_srvs_TARGETS}
  tf2::tf2
  tf2_ros::tf2_ros
)

# The messages are generated by this same package, so the typesupport target has to be resolved
# rather than found: there is no vr::vr_msgs to link against at configure time.
rosidl_get_typesupport_target(vr_msgs_cpp ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(vr_components PUBLIC "${vr_msgs_cpp}")

# Each component also gets a standalone executable, so one can be run alone while debugging.
rclcpp_components_register_node(vr_components
  PLUGIN "vr::SceneNode" EXECUTABLE scene_node)
rclcpp_components_register_node(vr_components
  PLUGIN "vr::InputNode" EXECUTABLE input_node)

# --- offline tool --------------------------------------------------------------------------
add_executable(scene_export src/scene_export.cpp)
target_link_libraries(scene_export PRIVATE vr_scene)

# MuJoCo ships as a plain .so outside the ament prefix; without this the installed binaries only
# run from a shell that exports LD_LIBRARY_PATH.
set_target_properties(vr_scene vr_components scene_export PROPERTIES
  BUILD_RPATH "${VR_MUJOCO_DIR}/lib"
  INSTALL_RPATH "${VR_MUJOCO_DIR}/lib"
)

# --- documentation -------------------------------------------------------------------------
# Off by default; the guides are readable as markdown without it. Requires: apt install doxygen
#   cmake --build build --target docs   ->  build/vr/docs/html/index.html
option(BUILD_DOCS "Generate HTML documentation with Doxygen" OFF)
if(BUILD_DOCS)
  # Older doxygen ships a div.fragment rule that outranks the theme's .fragment.
  find_package(Doxygen 1.16 REQUIRED)

  # The package lives in src/vr while the guides live at the workspace root, so Doxygen needs
  # both roots rather than assuming one tree.
  set(VR_DOXYGEN_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
  get_filename_component(VR_DOXYGEN_DOCS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

  set(VR_DOXYFILE "${CMAKE_CURRENT_BINARY_DIR}/Doxyfile")
  configure_file(docs/doxygen/Doxyfile.in "${VR_DOXYFILE}" @ONLY)

  add_custom_target(docs
    COMMAND ${DOXYGEN_EXECUTABLE} "${VR_DOXYFILE}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    COMMENT "Generating documentation with Doxygen"
    VERBATIM
  )
endif()

install(DIRECTORY include/ DESTINATION include)
install(TARGETS vr_scene vr_components
  EXPORT export_vr
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin
)
install(TARGETS scene_export DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

ament_export_targets(export_vr HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp geometry_msgs std_msgs)
ament_package()
