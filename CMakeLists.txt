cmake_minimum_required(VERSION 3.28)
# C is required: rosidl generates C sources for the message types.
project(vive_vr_ros2 C CXX)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rosidl_default_generators REQUIRED)

# ROS 2 has no gaze message, and PoseArray cannot say which body a pose belongs to.
rosidl_generate_interfaces(${PROJECT_NAME}
  msg/EyeGaze.msg
  msg/BodyPoses.msg
  DEPENDENCIES std_msgs geometry_msgs
)

find_package(mj_kdl_wrapper REQUIRED)

# The .glb embeds images as PNG, and PNG means deflate.
find_package(ZLIB REQUIRED)

# mujoco::mujoco comes from mj_kdl_wrapper's config: MuJoCo without its KDL, glfw and OpenGL,
# and guaranteed to be the copy the wrapper itself linked.
if(NOT TARGET mujoco::mujoco)
  message(FATAL_ERROR "mj_kdl_wrapper provides no mujoco::mujoco; update it and rebuild")
endif()
get_target_property(MUJOCO_LIBRARY mujoco::mujoco IMPORTED_LOCATION)
get_filename_component(MUJOCO_LIB_DIR "${MUJOCO_LIBRARY}" DIRECTORY)

add_library(vive_vr_core SHARED
  src/body_pose_publisher.cpp
  src/grabber.cpp
  src/gltf_writer.cpp
  src/geom_mesh.cpp
)
target_include_directories(vive_vr_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
# Generated by this same package, so there is no target to find at configure time.
rosidl_get_typesupport_target(vr_msgs_cpp ${PROJECT_NAME} rosidl_typesupport_cpp)

target_link_libraries(vive_vr_core PUBLIC
  mujoco::mujoco
  ZLIB::ZLIB
  rclcpp::rclcpp
  ${geometry_msgs_TARGETS}
  ${sensor_msgs_TARGETS}
  ${std_msgs_TARGETS}
  "${vr_msgs_cpp}"
)

add_library(vive_vr_components SHARED
  src/components/scene_node.cpp
  src/components/input_node.cpp
  src/components/teleop_node.cpp
)
target_link_libraries(vive_vr_components PUBLIC
  vive_vr_core
  mj_kdl_wrapper::mj_kdl_wrapper
  rclcpp::rclcpp
  rclcpp_components::component
  ${builtin_interfaces_TARGETS}
  ${geometry_msgs_TARGETS}
  ${sensor_msgs_TARGETS}
  ${std_msgs_TARGETS}
  ${std_srvs_TARGETS}
  tf2::tf2
  tf2_ros::tf2_ros
)

target_link_libraries(vive_vr_components PUBLIC "${vr_msgs_cpp}")

rclcpp_components_register_node(vive_vr_components
  PLUGIN "vive_vr_ros2::SceneNode" EXECUTABLE scene_node)
rclcpp_components_register_node(vive_vr_components
  PLUGIN "vive_vr_ros2::InputNode" EXECUTABLE input_node)
rclcpp_components_register_node(vive_vr_components
  PLUGIN "vive_vr_ros2::TeleopNode" EXECUTABLE teleop_node)

add_executable(scene_export src/scene_export.cpp)
target_link_libraries(scene_export PRIVATE vive_vr_core)

# MuJoCo ships as a plain .so outside the ament prefix; without this the installed binaries only
# run from a shell that exports LD_LIBRARY_PATH.
set_target_properties(vive_vr_core vive_vr_components scene_export PROPERTIES
  BUILD_RPATH "${MUJOCO_LIB_DIR}"
  INSTALL_RPATH "${MUJOCO_LIB_DIR}"
)

option(BUILD_DOCS "Generate HTML documentation with Doxygen" OFF)
if(BUILD_DOCS)
  add_subdirectory(docs)
endif()

# Off by default: the kitchen downloads gigabytes of RoboCasa assets, and a code build has no
# business needing the network.
option(BUILD_SCENES "Export the scenes in scenes/ into the cache" OFF)
add_custom_target(scenes
  COMMAND ${CMAKE_COMMAND} -E env python3 "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build_scenes.py"
          --scene-export "$<TARGET_FILE:scene_export>"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  COMMENT "Exporting scenes into the cache"
  VERBATIM
)
if(BUILD_SCENES)
  set_property(TARGET scenes PROPERTY EXCLUDE_FROM_ALL FALSE)
  add_custom_target(scenes_all ALL DEPENDS scenes)
endif()

install(DIRECTORY include/ DESTINATION include)
install(TARGETS vive_vr_core vive_vr_components
  EXPORT export_vive_vr_ros2
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin
)
install(TARGETS scene_export DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/vive_vr_discovery DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

ament_export_targets(export_vive_vr_ros2 HAS_LIBRARY_TARGET)

# Robot-specific simulation stays outside the reusable, MuJoCo-only VR core.
option(BUILD_OPENARM_SIM "Build the OpenArm simulation application" ON)
if(BUILD_OPENARM_SIM)
  add_library(openarm_sim_support SHARED src/openarm_ik.cpp src/openarm_teleop.cpp)
  target_link_libraries(openarm_sim_support PUBLIC vive_vr_core mj_kdl_wrapper::mj_kdl_wrapper)
  add_executable(openarm_sim_node src/openarm_sim_node.cpp)
  target_link_libraries(openarm_sim_node PRIVATE openarm_sim_support ${std_srvs_TARGETS})
  set_target_properties(openarm_sim_support openarm_sim_node PROPERTIES
    BUILD_RPATH "${MUJOCO_LIB_DIR}" INSTALL_RPATH "${MUJOCO_LIB_DIR}")
  install(TARGETS openarm_sim_support LIBRARY DESTINATION lib)
  install(TARGETS openarm_sim_node DESTINATION lib/${PROJECT_NAME})
endif()

if(BUILD_TESTING AND BUILD_OPENARM_SIM)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME openarm_launch_check
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/openarm_launch_check.py)
  add_test(NAME openarm_alignment_check
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/openarm_alignment_check.py)
  add_test(NAME openarm_alignment_capture_check
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/openarm_alignment_capture_check.py)
  set(OPENARM_TEST_MODEL "" CACHE FILEPATH "Generated OpenArm controlled MJCF for integration tests")
  foreach(check openarm_ik_check openarm_bridge_check)
    add_executable(${check} tests/${check}.cpp)
    target_link_libraries(${check} PRIVATE openarm_sim_support)
    if(OPENARM_TEST_MODEL)
      add_test(NAME ${check} COMMAND ${check} "${OPENARM_TEST_MODEL}")
    endif()
  endforeach()
endif()
ament_export_dependencies(rclcpp geometry_msgs std_msgs)
ament_package()
